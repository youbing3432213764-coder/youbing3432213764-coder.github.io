用file伪协议查看index.php,得到

<?php

error_reporting(0);

if (isset($_GET['N_S.S'])) {

    eval($_GET['N_S.S']);

}

if(!isset($_GET['file'])) {

    header('Location:/index.php?file=');

} else {

    $file = $_GET['file'];

    if (!preg_match('/\.\.|la|data|input|glob|global|var|dict|gopher|file|http|phar|localhost|\?|\*|\~|zip|7z|compress/is', $file)) {

        include $file;

    } else {

        die('error.');

    }

}
